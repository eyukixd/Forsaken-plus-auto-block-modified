-- Patch 198: Drag Block Helpless/cooldown handling.
-- Patch 199: Added WallCheck gate for Auto Block and Drag Block.
-- Physically walks toward the killer while Auto Block is active.
-- Helpless pauses an existing follow instead of resetting/cancelling it.
-- New Drag Block triggers are rejected during Helpless; cooldown time still advances.
-- All survivor movement inputs are blocked until the routine finishes.
local DragBlockBusy = false
local DragBlockToken = 0
local DragBlockLastFollowAt = -math.huge
-- Drag Block now follows the game's actual Block cooldown instead of a hardcoded timer.
-- The value is refreshed from the Block ability UI whenever the block is used.
local FOLLOW_KILLER_COOLDOWN = 0
local LastKnownBlockCooldown = 0
local BlockCooldownUntil = 0
local CancelDragBlockFollow = nil

-- Fail closed when the Block UI indicates an active cooldown. Drag Block must
-- never start merely because an attack was detected while Block is unavailable.
local function IsBlockCooldownActive(BlockButton)
    if not BlockButton then return true end
    if not BlockButton.Visible or not BlockButton.Active or BlockButton.Interactable == false then return true end
    if os.clock() < BlockCooldownUntil then return true end

    for _, AttributeName in {"CooldownRemaining", "RemainingCooldown", "CooldownLeft"} do
        local Remaining = tonumber(BlockButton:GetAttribute(AttributeName))
        if Remaining and Remaining > 0 then return true end
    end

    for _, Object in BlockButton:GetDescendants() do
        if not Object:IsA("GuiObject") or not Object.Visible then continue end
        local Name = string.lower(Object.Name or "")
        local IsCooldownObject = Name:find("cooldown", 1, true)
            or Name == "cd" or Name:find("countdown", 1, true)
            or Name:find("remaining", 1, true)
        if IsCooldownObject then
            if Object:IsA("TextLabel") or Object:IsA("TextButton") then
                local Remaining = tonumber(tostring(Object.Text or ""):match("(%d+%.?%d*)"))
                if Remaining and Remaining > 0 then return true end
            elseif Object:IsA("Frame") or Object:IsA("ImageLabel") or Object:IsA("ImageButton") then
                -- Only treat it as cooldown if it has a numeric countdown, or
                -- the button itself is disabled; plain decorative overlays vary by skin.
                for _, Child in Object:GetDescendants() do
                    if (Child:IsA("TextLabel") or Child:IsA("TextButton")) and Child.Visible then
                        local Remaining = tonumber(tostring(Child.Text or ""):match("(%d+%.?%d*)"))
                        if Remaining and Remaining > 0 then return true end
                    end
                end
            end
        end
    end
    return false
end

local function MarkBlockUsed(BlockButton)
    UpdateDragBlockCooldownFromBlockUI(BlockButton)
    local Duration = math.max(LastKnownBlockCooldown, FOLLOW_KILLER_COOLDOWN)
    if Duration > 0 then
        BlockCooldownUntil = math.max(BlockCooldownUntil, os.clock() + Duration)
    end
    task.delay(0.08, function()
        if BlockButton and BlockButton.Parent then
            UpdateDragBlockCooldownFromBlockUI(BlockButton)
            local UpdatedDuration = math.max(LastKnownBlockCooldown, FOLLOW_KILLER_COOLDOWN)
            if UpdatedDuration > 0 then
                BlockCooldownUntil = math.max(BlockCooldownUntil, os.clock() + UpdatedDuration)
            end
        end
    end)
end

-- Exact Helpless detection from ReplicatedStorage.Modules.Schematics.StatusEffects.Helpless.
-- The game's Applied handler sets Character.AbilitiesDisabled = true.
local function IsHelplessActive()
    local Character = LocalCharacter
    if not Character or not Character.Parent then
        return false
    end

    -- AbilitiesDisabled is the game's primary Helpless lock, but character/UI
    -- refreshes can briefly leave that attribute stale. Check the common status
    -- containers as a fallback so Auto Block and Drag Block fail closed.
    if Character:GetAttribute("AbilitiesDisabled") == true then
        return true
    end
    local Humanoid = Character:FindFirstChildOfClass("Humanoid")
    if Humanoid and Humanoid:GetAttribute("AbilitiesDisabled") == true then
        return true
    end
    for _, Object in Character:GetDescendants() do
        if string.lower(Object.Name) == "helpless" then
            if not Object:IsA("BoolValue") or Object.Value == true then
                return true
            end
        end
    end
    local StatusEffects = Character:FindFirstChild("StatusEffects") or Character:FindFirstChild("Statuses")
    if StatusEffects and (StatusEffects:FindFirstChild("Helpless") or StatusEffects:FindFirstChild("helpless")) then
        return true
    end
    return false
end

-- Patch 199: WallCheck prevents Auto Block and Drag Block from triggering
-- when a solid obstacle is between the survivor and the attacking killer.
-- It uses the same local/client-side raycast idea as the standalone tracer.
local function IsWallCheckBlocked(Killer)
    if not GetValue("WallCheck", true) then
        return false
    end

    local Character = LocalCharacter
    local Root = LocalRoot or (Character and Character:FindFirstChild("HumanoidRootPart"))
    if not Character or not Character.Parent or not Root or not Root.Parent
        or not Killer or not Killer.Parent then
        return false
    end

    local KillerRoot = Killer:FindFirstChild("HumanoidRootPart")
        or Killer:FindFirstChild("LowerTorso")
        or Killer:FindFirstChild("Torso")
    if not KillerRoot then
        return false
    end

    local Origin = Root.Position
    local Destination = KillerRoot.Position
    local Direction = Destination - Origin
    if Direction.Magnitude <= 0.05 then
        return false
    end

    local Params = RaycastParams.new()
    Params.FilterType = Enum.RaycastFilterType.Exclude
    Params.FilterDescendantsInstances = {Character, Killer}
    Params.IgnoreWater = true

    -- Skip non-solid/query-only decorative parts and keep checking until the
    -- line reaches the killer. Terrain or a solid collidable part counts as a wall.
    local CurrentOrigin = Origin
    local Remaining = Direction
    for _ = 1, 12 do
        local Result = workspace:Raycast(CurrentOrigin, Remaining, Params)
        if not Result then
            return false
        end

        local Hit = Result.Instance
        if Hit:IsA("Terrain") then
            return true
        end

        if Hit:IsA("BasePart") and Hit.CanCollide and Hit.Transparency < 1 then
            return true
        end

        local ToHit = Result.Position - CurrentOrigin
        local RemainingDistance = Remaining.Magnitude
        if ToHit.Magnitude >= RemainingDistance - 0.05 then
            return false
        end

        local Step = math.max(0.05, ToHit.Magnitude + 0.05)
        CurrentOrigin += Remaining.Unit * Step
        Remaining = Destination - CurrentOrigin
        if Remaining.Magnitude <= 0.05 then
            return false
        end
    end

    return true
end

-- Read the game's current Block cooldown from the Block ability UI.
-- Different game versions can expose it as an attribute or as a countdown label,
-- so support both without changing the game's Block cooldown itself.
local function UpdateDragBlockCooldownFromBlockUI(BlockButton)
    if not BlockButton then return end

    local function AcceptNumber(Value)
        local Number = tonumber(Value)
        if Number and Number > 0 and Number < 120 then
            LastKnownBlockCooldown = Number
            FOLLOW_KILLER_COOLDOWN = Number
            return true
        end
        return false
    end

    for _, AttributeName in {"Cooldown", "CooldownTime", "CooldownDuration", "Duration"} do
        if AcceptNumber(BlockButton:GetAttribute(AttributeName)) then
            return
        end
    end

    for _, Descendant in BlockButton:GetDescendants() do
        for _, AttributeName in {"Cooldown", "CooldownTime", "CooldownDuration", "Duration", "Time"} do
            if AcceptNumber(Descendant:GetAttribute(AttributeName)) then
                return
            end
        end

        if Descendant:IsA("TextLabel") or Descendant:IsA("TextButton") then
            local Name = string.lower(Descendant.Name or "")
            local Text = tostring(Descendant.Text or "")
            local Number = tonumber(Text:match("(%d+%.?%d*)"))
            if Number and (Name:find("cooldown", 1, true) or Name:find("cd", 1, true)) then
                if AcceptNumber(Number) then return end
            end
        end
    end
end

-- Normal Drag Block only.  Cooldown is recorded when a valid follow trigger
-- starts, so repeated ABH/sound callbacks cannot restart it early.
local function DragBlockAttackingKiller(Killer)
    if DragBlockBusy or not GetValue("DragBlock", false) then return end
    if IsWallCheckBlocked(Killer) then return end
    -- A new Drag Block may not start while Helpless is active.
    -- Existing Drag Block cooldowns continue normally during Helpless.
    if IsHelplessActive() then return end
    local Now = os.clock()
    local CurrentBlockCooldown = LastKnownBlockCooldown
    if CurrentBlockCooldown <= 0 then
        CurrentBlockCooldown = FOLLOW_KILLER_COOLDOWN
    end
    if CurrentBlockCooldown > 0 and Now - DragBlockLastFollowAt < CurrentBlockCooldown then return end
    if not Killer or not Killer.Parent then return end

    local Character = LocalCharacter
    local Humanoid = LocalHumanoid
    local Root = LocalRoot
    local KillerRoot = Killer:FindFirstChild("HumanoidRootPart")
    if not Character or not Humanoid or not Root or not KillerRoot then return end
    local ContextActionService = game:GetService("ContextActionService")
    local Players = game:GetService("Players")
    local MovementActionName = "AutoBlockFollowMovementOverride"
    local AbilityContainer = MainUI and MainUI:FindFirstChild("AbilityContainer")
    local BlockButton = AbilityContainer and AbilityContainer:FindFirstChild("Block")
    if not BlockButton or not BlockButton:IsA("GuiButton") then return end
    UpdateDragBlockCooldownFromBlockUI(BlockButton)
    if not BlockButton.Visible or not BlockButton.Active or BlockButton.Interactable == false
        or IsBlockCooldownActive(BlockButton) then
        return
    end
    -- Normal Drag Block uses the regular stop-distance setting.
    local KillerNameLower = string.lower(Killer.Name or "")
    local STOP_DISTANCE = tonumber(GetValue("DragBlockStopDistance")) or (KillerNameLower:find("c00lkidd", 1, true) and 2 or 3)
    -- Give MoveTo enough time to actually reach the configured target.
    -- The old 0.2s timeout could end the routine before any meaningful movement occurred.
    local HOLD_DURATION = tonumber(GetValue("DragBlockHoldDuration")) or 0.2
    local MOVE_TIMEOUT = math.max(1.5, HOLD_DURATION + 1.0)

    local function GetFlatDistance()
        if not Root or not Root.Parent or not KillerRoot or not KillerRoot.Parent then
            return math.huge
        end
        local Delta = Vector3.new(KillerRoot.Position.X - Root.Position.X, 0, KillerRoot.Position.Z - Root.Position.Z)
        return Delta.Magnitude
    end
    -- Start cooldown only after a valid follow trigger has been accepted.
    -- This timestamp cannot be shortened by later attack callbacks.
    DragBlockLastFollowAt = Now
    if GetFlatDistance() <= STOP_DISTANCE then return end

    DragBlockBusy = true
    DragBlockToken += 1
    local Token = DragBlockToken
    local OriginalWalkSpeed = Humanoid.WalkSpeed
    local OriginalJumpPower = Humanoid.JumpPower
    local OriginalJumpHeight = Humanoid.JumpHeight
    local OriginalAutoRotate = Humanoid.AutoRotate

    -- Disable Roblox's PlayerModule controls as well as sinking actions.
    -- This prevents the mobile thumbstick from mixing with Humanoid:MoveTo().
    local PlayerControls
    pcall(function()
        local PlayerScripts = Players.LocalPlayer:FindFirstChildOfClass("PlayerScripts")
        local PlayerModule = PlayerScripts and PlayerScripts:FindFirstChild("PlayerModule")
        if PlayerModule then
            local Controls = require(PlayerModule):GetControls()
            if Controls then
                PlayerControls = Controls
                PlayerControls:Disable()
            end
        end
    end)

    local function SinkMovementInput()
        return Enum.ContextActionResult.Sink
    end

    -- Sink keyboard, controller, mobile movement, jump, and autorun inputs.
    pcall(function()
        ContextActionService:BindActionAtPriority(
            MovementActionName,
            SinkMovementInput,
            false,
            Enum.ContextActionPriority.High.Value,
            Enum.PlayerActions.CharacterForward,
            Enum.PlayerActions.CharacterBackward,
            Enum.PlayerActions.CharacterLeft,
            Enum.PlayerActions.CharacterRight,
            Enum.PlayerActions.CharacterJump,
            Enum.PlayerActions.CharacterToggleAutoRun
        )
    end)

    -- Use Humanoid:MoveTo for real, physics-based walking toward the killer while manual movement inputs are blocked.
    Humanoid.WalkSpeed = 27
    Humanoid.JumpPower = 0
    Humanoid.JumpHeight = 0
    Humanoid.AutoRotate = false

    -- Smoothly rotate the survivor toward the attacking killer while Drag Block is active.
    -- The turn only changes rotation; movement toward the killer is still handled by Humanoid:MoveTo().
    local SmoothLookConnection
    local SmoothLookStarted = os.clock()
    local SMOOTH_LOOK_DURATION = 0.4
    local function StopSmoothLook()
        if SmoothLookConnection then
            SmoothLookConnection:Disconnect()
            SmoothLookConnection = nil
        end
    end

    -- There must be one cleanup path for both the render loop and the
    -- movement worker.  The render loop can reach the stop distance before
    -- MoveToFinished fires, and the worker can finish for another reason.
    local Finished = false
    local function FinishDragBlock()
        if Finished then return end
        Finished = true
        if Token == DragBlockToken then
            DragBlockToken += 1
        end
        StopSmoothLook()
        if Humanoid and Humanoid.Parent then
            pcall(function()
                if Root and Root.Parent then
                    -- Immediately cancel any active MoveTo target so the
                    -- survivor cannot continue walking after reaching the
                    -- configured Stop Near Killer distance.
                    Humanoid:MoveTo(Root.Position)
                    Humanoid:Move(Vector3.zero, true)
                    Root.AssemblyLinearVelocity = Vector3.zero
                    Root.AssemblyAngularVelocity = Vector3.zero
                end
            end)
            Humanoid.WalkSpeed = OriginalWalkSpeed
            Humanoid.JumpPower = OriginalJumpPower
            Humanoid.JumpHeight = OriginalJumpHeight
            Humanoid.AutoRotate = OriginalAutoRotate
            Humanoid:Move(Vector3.zero, true)
        end
        pcall(function()
            ContextActionService:UnbindAction(MovementActionName)
        end)
        pcall(function()
            if PlayerControls then
                PlayerControls:Enable()
            end
        end)
        DragBlockBusy = false
        if CancelDragBlockFollow == FinishDragBlock then
            CancelDragBlockFollow = nil
        end
    end
    CancelDragBlockFollow = FinishDragBlock

    SmoothLookConnection = RunService.RenderStepped:Connect(function(DeltaTime)
        if not DragBlockBusy
            or Token ~= DragBlockToken
            or not GetValue("DragBlock", false)
            or not Root
            or not Root.Parent
            or not KillerRoot
            or not KillerRoot.Parent then
            FinishDragBlock()
            return
        end

        -- Helpless pauses Drag Block without resetting/cancelling it.
        -- The follow resumes automatically when Helpless ends.
        if IsHelplessActive() then
            return
        end

        -- Keep checking for the stop distance for the whole follow, not just
        -- during the first 0.4 seconds of smooth rotation.
        if GetFlatDistance() <= STOP_DISTANCE then
            FinishDragBlock()
            return
        end
        if os.clock() - SmoothLookStarted >= SMOOTH_LOOK_DURATION then
            return
        end

        local LookPosition = Vector3.new(KillerRoot.Position.X, Root.Position.Y, KillerRoot.Position.Z)
        local LookDirection = LookPosition - Root.Position
        if LookDirection.Magnitude > 0.05 then
            local TargetCFrame = CFrame.lookAt(Root.Position, LookPosition)
            local SmoothAlpha = 1 - math.exp(-12 * math.max(DeltaTime, 0))
            Root.CFrame = Root.CFrame:Lerp(TargetCFrame, SmoothAlpha)
        end
    end)

    task.spawn(function()
        -- Walk toward the killer, stopping at the killer-specific distance.
        local direction = Vector3.new(KillerRoot.Position.X - Root.Position.X, 0, KillerRoot.Position.Z - Root.Position.Z)
        local distance = direction.Magnitude

        -- Use horizontal distance: different rig/root heights must not prevent
        -- the follow from stopping at the configured ground distance.
        if distance <= STOP_DISTANCE then
            FinishDragBlock()
            return
        end
        local TargetDistance = STOP_DISTANCE

        if distance > 0.05 then
            local startedMove = os.clock()
            while DragBlockBusy
                and Token == DragBlockToken
                and GetValue("DragBlock", false)
                and Humanoid.Health > 0
                and Killer.Parent
                and KillerRoot.Parent do
                -- Helpless pauses the active follow without cancelling it.
                -- The Drag Block cooldown remains independent and continues
                -- advancing normally because DragBlockLastFollowAt is not reset.
                if IsHelplessActive() then
                    task.wait(0.08)
                    continue
                end

                -- Only count time toward the movement timeout while the follow
                -- is actually allowed to run. Helpless time is not consumed.
                if os.clock() - startedMove >= MOVE_TIMEOUT then
                    break
                end

                local CurrentDirection = Vector3.new(KillerRoot.Position.X - Root.Position.X, 0, KillerRoot.Position.Z - Root.Position.Z)
                local CurrentDistance = CurrentDirection.Magnitude

                -- Force-end as soon as the stop distance is reached.  This is
                -- authoritative even if MoveTo has not reported completion.
                -- Do not wait for Hold Player Duration once the stop distance
                -- has been reached.
                if CurrentDistance <= STOP_DISTANCE then
                    FinishDragBlock()
                    return
                end
                if CurrentDistance > TargetDistance and CurrentDistance > 0.05 then
                    -- Aim at a point on the player's current ground plane.
                    local TargetPosition = Vector3.new(
                        KillerRoot.Position.X - CurrentDirection.Unit.X * TargetDistance,
                        Root.Position.Y,
                        KillerRoot.Position.Z - CurrentDirection.Unit.Z * TargetDistance
                    )
                    Humanoid:MoveTo(TargetPosition)
                else
                    FinishDragBlock()
                    return
                end

                task.wait(0.08)
            end
        end

        -- This also handles the timeout, a disabled setting, or a destroyed
        -- killer.  FinishDragBlock is idempotent if the render loop won the
        -- race and already stopped the follow.
        FinishDragBlock()
    end)
end

local function StopCurrentDragBlockFollow()
    local CancelFollow = CancelDragBlockFollow
    if CancelFollow then
        CancelFollow()
    end
end


-- Returns true when the local player is currently spectating.
-- Spectators do not run the Auto Do All Generators automation.
local function IsLocalPlayerSpectating()
    if LocalCharacter and LocalCharacter.Parent and LocalCharacter.Parent.Name == "Spectator" then
        return true
    end

    if LocalPlayer and LocalPlayer.Team and string.lower(LocalPlayer.Team.Name or "") == "spectator" then
        return true
    end

    return false
end

local function StopAutoDoAllInvincibility()
    AutoDoAllInvincibleToken += 1
    IsAutoDoAllGeneratorsRunning = false
    workspace:SetAttribute("Invincible", nil)
    local Invincible = GetValue("Invincible", true)
    if Invincible then
        Invincible.Value = false
        pcall(function()
            Invincible:GoUnder(false)
        end)
    end

    local AutoPuzzle = GetValue("AutoGeneratorPuzzle", true)
    if AutoDoAllPuzzleWasEnabled == false and AutoPuzzle then
        AutoPuzzle.Value = false
    end
    AutoDoAllPuzzleWasEnabled = nil
end

local function EnableAutoDoAllPuzzleAssist()
    local AutoPuzzle = GetValue("AutoGeneratorPuzzle", true)
    if not AutoPuzzle then
        return
    end
    if AutoDoAllPuzzleWasEnabled == nil then
        AutoDoAllPuzzleWasEnabled = AutoPuzzle.Value == true
    end
    if not AutoPuzzle.Value then
        AutoPuzzle.Value = true
    end
end

-- Finds a clear interaction position around a generator.
-- This prevents another survivor standing on the preferred side
-- from blocking the normal generator interaction.
local function GetClearGeneratorPosition(GeneratorMain)
    local Root = LocalRoot or (LocalCharacter and LocalCharacter:FindFirstChild("HumanoidRootPart"))
    if not GeneratorMain or not Root then
        return nil
    end

    local GeneratorCFrame = GeneratorMain.CFrame
    local GeneratorSize = GeneratorMain.Size
    local CheckDistance = math.max(4.5, math.max(GeneratorSize.X, GeneratorSize.Z) * 0.5 + 3)

    -- Only use the two sides and the front. Never use the back.
    local Positions = {
        GeneratorCFrame.Position + GeneratorCFrame.RightVector * CheckDistance,
        GeneratorCFrame.Position - GeneratorCFrame.RightVector * CheckDistance,
        GeneratorCFrame.Position + GeneratorCFrame.LookVector * CheckDistance
    }

    local RaycastParams = RaycastParams.new()
    RaycastParams.FilterType = Enum.RaycastFilterType.Exclude
    RaycastParams.FilterDescendantsInstances = {LocalCharacter, GeneratorMain.Parent}
    local FallbackPosition = Positions[1]

    for _, Position in Positions do
        local Clear = true

        -- Make sure the player's body would not overlap solid geometry.
        local OccupyingParts = workspace:GetPartBoundsInBox(
            CFrame.new(Position),
            Vector3.new(3, 5, 3),
            RaycastParams
        )

        for _, Part in ipairs(OccupyingParts) do
            if Part.CanCollide then
                Clear = false
                break
            end
        end

        if Clear then
            for _, Player in Players:GetPlayers() do
                if Player ~= LocalPlayer and Player.Character then
                    local OtherRoot = Player.Character:FindFirstChild("HumanoidRootPart")
                    if OtherRoot and (OtherRoot.Position - Position).Magnitude < 4 then
                        Clear = false
                        break
                    end
                end
            end
        end

        if Clear then
            -- Make sure there is no wall directly between the generator and
            -- the interaction position.
            local Direction = Position - GeneratorCFrame.Position
            local WallCheck = workspace:Raycast(
                GeneratorCFrame.Position,
                Direction.Unit * Direction.Magnitude,
                RaycastParams
            )

            if WallCheck then
                Clear = false
            end
        end

        if Clear then
            return Position
        end
    end

    -- A blocked test should not prevent the automation from trying the
    -- generator. The prompt/interaction retry logic handles the fallback.
    return FallbackPosition
end

            ["HandlePuzzle"] = function(Data)
                PuzzleData = {}
                local function GetDirection(Row1,Col1,Row2,Col2)
                    if Row2 < Row1 then
                        return "up"
                    end
                    if Row2 > Row1 then
                        return "down"
                    end
                    if Col2 < Col1 then
                        return "left"
                    end
                    if Col2 > Col1 then
                        return "right"
                    end
                end
                local function GetPuzzleConnections(Last, New, Next)
                    local Connections = {}
                    if Last and New then
                        local Direction = GetDirection(New.row, New.col, Last.row, Last.col)
                        if Direction == "up" then
                            Direction = "down"
                        elseif Direction == "down" then
                            Direction = "up"
                        elseif Direction == "left" then
                            Direction = "right"
                        elseif Direction == "right" then
                            Direction = "left"
                        end
                        if Direction ~= "" and Direction then
                            Connections[Direction] = true
                        end
                    end
                    if Next and New then
                        local Direction = GetDirection(New.row, New.col, Next.row, Next.col)
                        if Direction ~= "" and Direction then
                            Connections[Direction] = true
                        end
                    end
                    return Connections
                end
                local function NearbyDirection(Row1, Col1, Row2, Col2)
                    if Row2 == Row1 - 1 and Col2 == Col1 then
                        return "up"
                    end
                    if Row2 == Row1 + 1 and Col2 == Col1 then
                        return "down"
                    end
                    if Row2 == Row1 and Col2 == Col1 - 1 then
                        return "left"
                    end
                    if Row2 == Row1 and Col2 == Col1 + 1 then
                        return "right"
                    end
                    return false
                end
                local function GetPlaceKey(Point)
                    return `{Point.row}-{Point.col}`
                end
                local function PathEndpoints(Path, endpoints)
                    if not Path or #Path == 0 then
                        return Path
                    end
                    local UniquePoint
                    for i, Point in endpoints or {} do
                        for i, v in Path do
                            if v.row == Point.row and v.col == Point.col then
                                UniquePoint = {row = Point.row,col = Point.col}
                                break
                            end
                        end
                        if UniquePoint then
                            break
                        end
                    end
                    if not UniquePoint then
                        local PointsInPath = {}
                        for i, Point in Path do
                            local TableKey = GetPlaceKey(Point)
                            PointsInPath[TableKey] = Point
                        end
                        for i, Point in Path do
                            local Nearby = 0
                            local Directions = {{Point.row - 1,Point.col}, {Point.row + 1,Point.col},{Point.row,Point.col - 1}, {Point.row,Point.col + 1}}
                            for i, Data in Directions do
                                local Row = Data[1]
                                local Col = Data[2]
                                local TableKey = `{Row}-{Col}`
                                if PointsInPath[TableKey] ~= nil then
                                    Nearby += 1
                                end
                            end
                            if Nearby == 1 then
                                UniquePoint = {row = Point.row,col = Point.col}
                                break
                            end
                        end
                    end
                    if not UniquePoint then
                        UniquePoint = {row = Path[1].row,col = Path[1].col}
                    end
                    local Remaining = {}
                    for i,Point in Path do
                        local TableKey = GetPlaceKey(Point)
                        local CurrentTableKey = {row = Point.row,col = Point.col}
                        Remaining[TableKey] = CurrentTableKey
                    end
                    local Ordered = {}
                    local CurrentTableKey = {row = UniquePoint.row,col = UniquePoint.col}
                    local ClonedTable = table.clone(CurrentTableKey)
                    setmetatable(ClonedTable, nil)
                    table.insert(Ordered, ClonedTable)
                    local TableKey = GetPlaceKey(CurrentTableKey)
                    Remaining[TableKey] = nil
                    while true do
                        local Size = 0
                        for i in Remaining do
                            Size += 1
                        end
                        if not (Size > 0) then
                            break
                        end
                        local foundNext = false
                        for key, Point in Remaining do
                            local Direction = NearbyDirection(CurrentTableKey.row,CurrentTableKey.col,Point.row,Point.col)
                            if Direction ~= "" and Direction then
                                local ClonedPoint = table.clone(Point)
                                setmetatable(ClonedPoint, nil)
                                table.insert(Ordered, ClonedPoint)
                                Remaining[key] = nil
                                CurrentTableKey = Point
                                foundNext = true
                                break
                            end
                        end
                        if not foundNext then
                            return Path
                        end
                    end
                    return Ordered
                end

            ["HandleNoliNPC"] = function(self,Value)
                -- Keep hidden Noli clones out of collision/query/physics work.
                -- Moving the model to Lighting alone does not make its BaseParts
                -- non-physical, so the clone could continue participating in
                -- engine spatial/physics processing while hidden.
                self.NoliNPCFrozenState = self.NoliNPCFrozenState or setmetatable({}, {__mode = "k"})

                local function FreezeHiddenNoli(Model)
                    if not Model or Players:GetPlayerFromCharacter(Model) then return end
                    if not self.NoliNPCFrozenState[Model] then
                        local State = {Parts = {}, Humanoids = {}}
                        for _,Descendant in Model:GetDescendants() do
                            if Descendant:IsA("BasePart") then
                                State.Parts[Descendant] = {
                                    Anchored = Descendant.Anchored,
                                    CanCollide = Descendant.CanCollide,
                                    CanTouch = Descendant.CanTouch,
                                    CanQuery = Descendant.CanQuery,
                                    Massless = Descendant.Massless
                                }
                                Descendant.Anchored = true
                                Descendant.CanCollide = false
                                Descendant.CanTouch = false
                                Descendant.CanQuery = false
                                Descendant.Massless = true
                            elseif Descendant:IsA("Humanoid") then
                                State.Humanoids[Descendant] = {AutoRotate = Descendant.AutoRotate}
                                Descendant.AutoRotate = false
                            end
                        end
                        self.NoliNPCFrozenState[Model] = State
                    else
                        for Part in pairs(self.NoliNPCFrozenState[Model].Parts) do
                            if Part and Part.Parent then
                                Part.Anchored = true
                                Part.CanCollide = false
                                Part.CanTouch = false
                                Part.CanQuery = false
                                Part.Massless = true
                            end
                        end
                    end
                end

                local function RestoreNoli(Model)
                    local State = self.NoliNPCFrozenState and self.NoliNPCFrozenState[Model]
                    if not State then return end

                        if Part and Part.Parent then
                            Part.Anchored = Original.Anchored
                            Part.CanCollide = Original.CanCollide
                            Part.CanTouch = Original.CanTouch
                            Part.CanQuery = Original.CanQuery
                            Part.Massless = Original.Massless
                        end
                    end
                    for Humanoid,Original in pairs(State.Humanoids) do
                        if Humanoid and Humanoid.Parent then
                            Humanoid.AutoRotate = Original.AutoRotate
                        end
                    end
                    self.NoliNPCFrozenState[Model] = nil
                end

-- Patch 168: Always read the current ABH NumberValue from the settings folder.
-- Some menu implementations can replace the NumberValue when a slider changes,
-- which would leave the cached Values[] entry pointing at the old value.
function GetCurrentHitboxAdjustment(FeatureName)
    local CurrentInstance = PlusFolderSettings and PlusFolderSettings:FindFirstChild(FeatureName, true)
    if CurrentInstance and CurrentInstance:IsA("NumberValue") then
        return tonumber(CurrentInstance.Value) or 0
    end

    -- Never fall back to Values[] here: a replaced slider instance must not
    -- leave an old individual-killer ABH adjustment active.
    return 0
end

function RichTextGradientColor(Text:string,Colors)
	local Count = 0
    for Char in Text:gmatch(utf8.charpattern) do
        if Char ~= " " then
			Count += 1
		end
	end
	if Count == 0 then return Text end
	local MaxIndex = math.max(1,Count-1)
	local Segments = #Colors-1
	local Result,Used = "",0
    for Char in Text:gmatch(utf8.charpattern) do
		if Char == " " then
			Result..=Char
		elseif Char:byte() >= 128 then
		else
			local T = Used/MaxIndex
			Used += 1
			local Segment= math.min(math.floor(T*Segments),Segments-1)
			Result..=('<font color="#%s">%s</font>'):format(Colors[Segment+1]:Lerp(Colors[Segment+2],T*Segments-Segment):ToHex(),Char)
		end
	end
	return Result
end

local function collidewith(gui1,gui2)
	local gui1_topLeft,gui1_bottomRight = gui1.AbsolutePosition, gui1.AbsolutePosition + gui1.AbsoluteSize
	local gui2_topLeft,gui2_bottomRight = gui2.AbsolutePosition, gui2.AbsolutePosition + gui2.AbsoluteSize
	return ((gui1_topLeft.x < gui2_bottomRight.x and gui1_bottomRight.x > gui2_topLeft.x) and (gui1_topLeft.y < gui2_bottomRight.y and gui1_bottomRight.y > gui2_topLeft.y))
end

function TableValueFind(Table, MatchFn, Seen)
    if type(Table) ~= "table" or type(MatchFn) ~= "function" then
        return nil
    end
    Seen = Seen or {}
    if Seen[Table] then
        return nil
    end
    Seen[Table] = true

    for Key, Value in Table do
        if MatchFn(Key, Value) then
            return Value, Key, Table
        elseif type(Value) == "table" then
            local FoundValue, FoundKey, FoundParent = TableValueFind(Value, MatchFn, Seen)
            if FoundKey ~= nil then
                return FoundValue, FoundKey, FoundParent
            end
        end
    end
    return nil
end

function GetAnimationType(ID)
    for i,v in AllAnimations do
        for animtype,animId in v do
            if type(animId) == "table" then
                for i2,v2 in animId do
                    if type(v2) == "string" and v2:find(tostring(ID)) then
                        return animtype,i,i2
                    end
                end
            else
                if type(animId) == "string" and animId:find(tostring(ID)) then
                    return animtype,i
                end
            end
        end
    end
end

local function AddOverridenAnimation(ID)
    if LocalCharacter and ID then
        if OverriddenAnimations[ID] then
            return table.unpack(OverriddenAnimations[ID])
        end
        local OverrideFolder = LocalCharacter:FindFirstChild("OverrideAnimation") or Instance.new("Folder", LocalCharacter)
        OverrideFolder.Name = "OverrideAnimation"
        local AnimType,CharName = GetAnimationType(ID) or "Unknown"
        local Animation = Instance.new("Animation")
        Animation.Name = AnimType .. tostring(ID)
        Animation.AnimationId = tostring(ID):find("id") and tostring(ID) or "http://www.roblox.com/asset/?id=" .. tostring(ID)
        Animation.Parent = OverrideFolder
        Animation:SetAttribute("Overriden",true)
        if LocalAnimator then
            local Track = LocalAnimator:LoadAnimation(Animation)
            OverriddenAnimations[ID] = {Track,Animation}
            return Track,Animation
        end
    else
        return
    end
end

local SupportedOverrides = {"idle","walk","run"}
function UpdateAnim(Humanoid)
    local AnimSelected = GetValue("AnimationChanger")
    if not CanPlayOverrideAnim(LocalCharacter) then return end
    local AnimString
    local Type
    if MainModule["IsSprinting"] and Humanoid.MoveDirection ~= Vector3.zero then
        AnimString = AllAnimations[AnimSelected] and AllAnimations[AnimSelected]["Run"]
        Type = "Run"
    elseif Humanoid.MoveDirection ~= Vector3.zero then
        AnimString = AllAnimations[AnimSelected] and AllAnimations[AnimSelected]["Walk"]
        Type = "Walk"
    else
        AnimString = AllAnimations[AnimSelected] and AllAnimations[AnimSelected]["Idle"]
        Type = "Idle"
    end
    if AnimString then
        local FoundAction = false
        for i,v in Humanoid:GetPlayingAnimationTracks() do
            if v.Animation and v.Animation:GetAttribute("Overriden") then
                v:Stop(0.2)
                continue
            end
            if not v.IsPlaying then
                continue
            end
                local AnimType,Char,Name = GetAnimationType(v.Animation.AnimationId)
                local IsSupported = false
                for i,v in SupportedOverrides do
                    if AnimType and AnimType:lower():find(v:lower()) then
                        IsSupported = true
                    end
                end
                if not IsSupported then
                    FoundAction = v
                end
        end
        local OverrideTrack,Animation = AddOverridenAnimation(AnimString)
        if OverrideTrack and CanPlayOverrideAnim(LocalCharacter) then
            OverrideTrack:Play(not FoundAction and 0.1 or 0.3)
        end
    end
end

        LocalAnimator.AnimationPlayed:Connect(function(Track)
            if CanPlayOverrideAnim(LocalCharacter) and MainModule then
                local Override = Track.Animation:GetAttribute("Overriden")
                if Override then 
                    --return
                end
                LastAnimOriginalUsed = not Override and Track or LastAnimOriginalUsed
                local AnimType,CharName = GetAnimationType(tonumber(Track.Animation.AnimationId:match("%d+")))
                if not Override then
                    for i,v in SupportedOverrides do
                        if AnimType and AnimType:lower():find(v:lower()) then
                            Track:Stop(0)
                        elseif not AnimType and Track.Looped then
                            Track:Stop(0)
                        end
                    end
                end
                local FoundAction = false
                for i,v in LocalAnimator:GetPlayingAnimationTracks() do
                    task.spawn(function()
                        local AnimType,Char,Name = GetAnimationType(v.Animation.AnimationId)
                        local IsSupported = false
                        for i,v in SupportedOverrides do
                            if AnimType and AnimType:lower():find(v:lower()) then
                                IsSupported = true
                            end
                        end
                        if v.IsPlaying and not IsSupported and not v:GetAttribute("Ignored") then
                            FoundAction = v
                        end
                    end)
                end
                if FoundAction and Override then
                    Track:Stop(0.2)
                    task.wait(not FoundAction.Looped and FoundAction.Length * 0.95 or 0)
                    if FoundAction.Looped then
                        FoundAction.Stopped:Wait()
                    end
                    FoundAction:SetAttribute("Ignored",true)
                    task.delay(0.2,function()
                        FoundAction:SetAttribute("Ignored",nil)
                    end)
                    UpdateAnim(LocalHumanoid)
                end
            end
        end)
        local MoreThanZero = 0
        LocalHumanoid:GetPropertyChangedSignal("MoveDirection"):Connect(function()
            if GetValue("AnimationChanger") ~= "Original" and MoreThanZero ~= (LocalHumanoid.MoveDirection.Magnitude == 0) then
                MoreThanZero = (LocalHumanoid.MoveDirection.Magnitude == 0)
                UpdateAnim(LocalHumanoid)
            end
        end)
        task.delay(0.05,UpdateAnim,LocalHumanoid)
    end)
end

local function HandleUIFeatures(TabName,TabContents)
    local Folder = Instance.new("Folder")
    Folder.Name = TabName
    Folder.Parent = PlusFolderSettings
    if not TabContents["TabAttributes"] then
        Folder:Destroy()
        warn(`Failed to create section : {TabName}\nERROR: There is no TabAttributes attached to the section`)
        return
    end
    for Attribute, Value in TabContents["TabAttributes"] do
        Folder:SetAttribute(Attribute, Value)
    end
    for SettingName, SettingData in TabContents do
        task.spawn(function()
            if SettingName == "TabAttributes" then
                return
            end
            if type(SettingData) ~= "table" or type(SettingData.InstanceType) ~= "string" then
                return
            end
            local NewInstance = Instance.new(SettingData.InstanceType)
            local TableEvent = table.clone(SettingData)
            table.clear(SettingData)
            setmetatable(SettingData,{
                __index = TableEvent,
                __newindex = function(Table,i,v)
                    TableEvent[i] = v
                    if i == "DisplayDescription" or i == "DisplayTitle" or i == "LayoutOrder" then
                        NewInstance:SetAttribute("DisplayDescription",type(TableEvent["DisplayDescription"]) == type("") and TableEvent["DisplayDescription"] or "")
                        NewInstance:SetAttribute("DisplayTitle",type(TableEvent["DisplayTitle"]) == type("") and TableEvent["DisplayTitle"] or "Untitled Feature")
                        NewInstance:SetAttribute("LayoutOrder",type(TableEvent["LayoutOrder"]) == type(1) and TableEvent["LayoutOrder"] or #NewInstance:GetAttribute("DisplayTitle"))
                    end
                    if i == "Requirement" then
                        NewInstance:SetAttribute("Requirement",(type(TableEvent["Requirement"]) == type(true) or TableEvent["Requirement"] == type("") and TableEvent["Requirement"]))
                    end
                end
            })
            NewInstance.Name = SettingName
            NewInstance.Value = SettingData.DefaultInstanceValue
            local TableData = HttpService:JSONDecode(ReturnData("Data.txt", true))
            if SettingData["Savable"] and FeatureLoadout["EnviromentFunctions"]["files"]["DefaultInstanceValue"] then
                if TableData[SettingName] ~= nil and NewInstance:GetAttribute("Requirement") ~= true then
                    NewInstance.Value = TableData[SettingName]
                end
                NewInstance:GetPropertyChangedSignal("Value"):Connect(function()
                    if NewInstance:GetAttribute("Requirement") == true or not SettingData["Savable"] then
                        return
                    end
                    local TableData2 = HttpService:JSONDecode(ReturnData("Data.txt", true))
                    TableData2[SettingName] = SettingData.DefaultInstanceValue == NewInstance.Value and nil or NewInstance.Value
                    if FeatureLoadout["EnviromentFunctions"]["files"]["DefaultInstanceValue"] then
                        ChangeData("Data.txt",HttpService:JSONEncode(TableData2),true)
                    end
                end)
            elseif not FeatureLoadout["EnviromentFunctions"]["files"]["DefaultInstanceValue"] and SettingData["Savable"] and not WarnedAboutFilesCompatability then
                WarnedAboutFilesCompatability = true
                ColoredPrint("Failed to load a savable feature.\nIf this bothers you then you should use a different executor that supports 'writefile' and 'readfile'", "info", Color3.new(1,0.25,0))
            end
            if SettingData["DisplayDescription"] then
                SettingData.DisplayDescription = SettingData.DisplayDescription:gsub("%(NO FUNCTIONALITY%)", RichTextGradientColor("(NO FUNCTIONALITY)",{Color3.fromRGB(255, 83, 53), Color3.fromRGB(255, 123, 0),Color3.fromRGB(255, 83, 53)}))
            end
            SettingData["Instance"] = NewInstance
            NewInstance:SetAttribute("DisplayDescription",SettingData["DisplayDescription"] or "")
            NewInstance:SetAttribute("DisplayTitle",SettingData["DisplayTitle"] or "Untitled Feature")
            NewInstance:SetAttribute("LayoutOrder",SettingData["LayoutOrder"] or #NewInstance:GetAttribute("DisplayTitle"))
            NewInstance:SetAttribute("Requirement",SettingData["Requirement"])
            if SettingData["ExtraData"] then
                for ExtraDataIndex, ExtraDataValue in SettingData.ExtraData do
                    NewInstance:SetAttribute(ExtraDataIndex, ExtraDataValue)
                end
            end
            NewInstance.Parent = Folder
            SetupSettingUI(NewInstance,Folder,SettingData)
        end)
    end
    SetupSettingUI(Folder)
    return Folder
end

local function UICheck()
    local Status = SideBar:GetAttribute("WasVisible")
    SetButtonState(false)
    if Status then
        if not NewUIVersion then
            local LayoutInstance = Buttons:FindFirstChildOfClass("UIListLayout") or Buttons:FindFirstChildOfClass("UIGridLayout")
            if LayoutInstance then
                LayoutInstance.VerticalAlignment = Enum.VerticalAlignment.Top
            end
        else
            SideBar:FindFirstChildOfClass("UIPadding").PaddingRight = UDim.new(0.2,0)
            SideBar:FindFirstChildOfClass("UIPadding").PaddingLeft = UDim.new(0,0)
            UIScale.Scale = 1
        end
        PlusMenu.SettingsContainer.ImageTransparency = 0
        PlusMenu.SettingsContainer.Contents.ScrollBarImageTransparency = 0
        if SideBar:IsA("ImageLabel") then
            SideBar.ImageTransparency = 0
        end
        for i,v in PlusMenu.SettingsContainer.Contents:GetChildren() do
            if v:IsA("ImageLabel") then
                v.ImageTransparency = 0
            elseif v:IsA("Frame") then
                for i,v in v:GetChildren() do
                    if v:IsA("ImageLabel") or v:IsA("ImageButton") then
                        v.ImageTransparency = 0
                    end
                end
            end
        end
    else
        if NewUIVersion then
            MenuData.SidebarMenus["Plus"]["_sidebar"]["MenusHidden"] = false
            task.delay(0.5,function()
                MenuData.SidebarMenus["Plus"]["_sidebar"]["MenusHidden"] = false
            end)
            SideBar:FindFirstChildOfClass("UIPadding").PaddingRight = UDim.new(0,0)
            SideBar:FindFirstChildOfClass("UIPadding").PaddingLeft = UDim.new(0.2,0)
            UIScale.Scale = 1.3
        end
        local LayoutInstance = Buttons:FindFirstChildOfClass("UIListLayout") or Buttons:FindFirstChildOfClass("UIGridLayout")
        if LayoutInstance then
            LayoutInstance.VerticalAlignment = Enum.VerticalAlignment.Center
        end
        PlusMenu.SettingsContainer.ImageTransparency = 0.3
        PlusMenu.SettingsContainer.Contents.ScrollBarImageTransparency = 0.45
        if SideBar:IsA("ImageLabel") then
            SideBar.ImageTransparency = 1
        end
        for i,v in PlusMenu.SettingsContainer.Contents:GetChildren() do
            if v:IsA("ImageLabel") then
                v.ImageTransparency = 0.4
            elseif v:IsA("Frame") then
                for i,v in v:GetChildren() do
                    if v:IsA("ImageLabel") or v:IsA("ImageButton") then
                        v.ImageTransparency = 0.3
                    end
                end
            end
        end
    end
    for i,v in SideBar:QueryDescendants(`#Bottom > Frame, #Money , #Bottombar , #{Buttons} > Frame {((not NewUIVersion) and ":not(#Credits)" or "")} :not(#TesterUi)`) do
        if not v:GetAttribute("Visibility") then
            v:SetAttribute("Visibility",v.Visible)
        end
        v.Visible = Status and v:GetAttribute("Visibility")
    end
    SidePlusButton.Visible = true
    if not Status then
        RoundEvent:Fire("Start")
    else
        RoundEvent:Fire("End")
    end
end

    AutoBlockToggleGui.IgnoreGuiInset = true
    AutoBlockToggleGui.Parent = CoreGui

    local AutoBlockToggle = Instance.new("TextButton")
    AutoBlockToggle.Name = "Toggle"
    AutoBlockToggle.Parent = AutoBlockToggleGui
    AutoBlockToggle.Size = UDim2.new(0, 150, 0, 48)
    AutoBlockToggle.Position = UDim2.new(1, -170, 0.5, -24)
    AutoBlockToggle.BackgroundTransparency = 0.15
    AutoBlockToggle.BorderSizePixel = 0
    AutoBlockToggle.TextColor3 = Color3.new(1, 1, 1)
    AutoBlockToggle.TextScaled = true
    AutoBlockToggle.Font = Enum.Font.GothamBold
    AutoBlockToggle.ZIndex = 100
    AutoBlockToggle.Active = true
    AutoBlockToggle.Selectable = true

    local Corner = Instance.new("UICorner")
    Corner.CornerRadius = UDim.new(0, 10)
    Corner.Parent = AutoBlockToggle

    local Stroke = Instance.new("UIStroke")
    Stroke.Thickness = 2
    Stroke.Parent = AutoBlockToggle

    local AutoBlockValue = GetValue("AutoBlock", true)
    local function UpdateAutoBlockToggle()
        local Enabled = AutoBlockValue and AutoBlockValue.Value or false
        AutoBlockToggle.Text = Enabled and "Auto Block: ON" or "Auto Block: OFF"
        AutoBlockToggle.BackgroundColor3 = Enabled and Color3.fromRGB(0, 170, 85) or Color3.fromRGB(45, 45, 45)
    end

    AutoBlockToggle.MouseButton1Click:Connect(function()
        if AutoBlockValue then
            AutoBlockValue.Value = not AutoBlockValue.Value
            UpdateAutoBlockToggle()
        end
    end)

    if AutoBlockValue then
        AutoBlockValue:GetPropertyChangedSignal("Value"):Connect(UpdateAutoBlockToggle)
    end
    UpdateAutoBlockToggle()

    -- Drag ONLY from the left/right side handles.
    -- The center of the button remains click-only, which prevents
    -- two-finger touches from fighting over the toggle.
    local Dragging = false
    local DragStart
    local StartPosition

    local function MakeDragHandle(Name, XPosition)
        local Handle = Instance.new("Frame")
        Handle.Name = Name
        Handle.Parent = AutoBlockToggle
        Handle.Size = UDim2.new(0, 28, 1, 0)
        Handle.Position = UDim2.new(XPosition, 0, 0, 0)
        Handle.BackgroundTransparency = 1
        Handle.BorderSizePixel = 0
        Handle.Active = true
        Handle.ZIndex = 101

        Handle.InputBegan:Connect(function(Input)
            if Input.UserInputType == Enum.UserInputType.MouseButton1
                or Input.UserInputType == Enum.UserInputType.Touch then

                Dragging = true
                DragStart = Input.Position
                StartPosition = AutoBlockToggle.Position

                Input.Changed:Connect(function()
                    if Input.UserInputState == Enum.UserInputState.End then
                        Dragging = false
                    end
                end)
            end
        end)

        return Handle
    end

    MakeDragHandle("LeftDragHandle", 0)
    MakeDragHandle("RightDragHandle", 1)

    UserInputService.InputChanged:Connect(function(Input)
        if Dragging and (Input.UserInputType == Enum.UserInputType.MouseMovement
            or Input.UserInputType == Enum.UserInputType.Touch) then

            local Delta = Input.Position - DragStart
            AutoBlockToggle.Position = UDim2.new(
                StartPosition.X.Scale, StartPosition.X.Offset + Delta.X,
                StartPosition.Y.Scale, StartPosition.Y.Offset + Delta.Y
            )
        end
    end)
end
